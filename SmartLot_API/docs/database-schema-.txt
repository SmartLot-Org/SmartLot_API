-- reservas

create table public.reservas (
  id serial not null,
  id_usuario integer null,
  id_garage integer null,
  id_vehiculo integer null,
  fecha_entrada timestamp without time zone null,
  fecha_salida timestamp without time zone null,
  entro boolean null default false,
  salio boolean null default false,
  "Borrado" boolean not null default false,
  dia public.dia_semana null,
  id_trato bigint null,
  modalidad_pago_aplicada public.modalidad_pago_trato_enum not null default 'empresa_cubre_cupo'::modalidad_pago_trato_enum,
  tipo_cupo public.tipo_cupo_reserva_enum not null default 'dentro_cupo'::tipo_cupo_reserva_enum,
  responsable_pago public.responsable_pago_enum not null default 'empresa'::responsable_pago_enum,
  tarifa_hora_aplicada numeric(12, 2) null,
  importe_estimado numeric(12, 2) null,
  estado_reserva public.estado_reserva_enum not null default 'confirmada'::estado_reserva_enum,
  retencion_pago_hasta timestamp with time zone null,
  qr_token uuid not null default gen_random_uuid (),
  qr_salida_token uuid null,
  constraint reservas_pkey primary key (id),
  constraint reservas_id_trato_fkey foreign KEY (id_trato) references trato_empresa_garage (id) on delete RESTRICT,
  constraint reservas_id_usuario_fkey foreign KEY (id_usuario) references usuarios (id),
  constraint reservas_id_vehiculo_fkey foreign KEY (id_vehiculo) references vehiculos (id),
  constraint reservas_id_garage_fkey foreign KEY (id_garage) references garages (id),
  constraint reservas_empresa_sin_retencion_check check (
    (
      (
        responsable_pago <> 'empresa'::responsable_pago_enum
      )
      or (retencion_pago_hasta is null)
    )
  ),
  constraint reservas_tarifa_hora_aplicada_check check (
    (
      (tarifa_hora_aplicada is null)
      or (tarifa_hora_aplicada >= (0)::numeric)
    )
  ),
  constraint reservas_importe_estimado_check check (
    (
      (importe_estimado is null)
      or (importe_estimado >= (0)::numeric)
    )
  ),
  constraint reservas_modalidad_responsable_check check (
    (
      (
        (
          modalidad_pago_aplicada = 'empresa_cubre_cupo'::modalidad_pago_trato_enum
        )
        and (tipo_cupo = 'dentro_cupo'::tipo_cupo_reserva_enum)
        and (
          responsable_pago = 'empresa'::responsable_pago_enum
        )
      )
      or (
        (
          modalidad_pago_aplicada = 'empresa_cubre_cupo'::modalidad_pago_trato_enum
        )
        and (tipo_cupo = 'extra'::tipo_cupo_reserva_enum)
        and (
          responsable_pago = 'empleado'::responsable_pago_enum
        )
      )
      or (
        (
          modalidad_pago_aplicada = 'empleado_paga_todo'::modalidad_pago_trato_enum
        )
        and (
          responsable_pago = 'empleado'::responsable_pago_enum
        )
      )
    )
  ),
  constraint reservas_qr_salida_distinto_chk check (
    (
      (qr_salida_token is null)
      or (qr_salida_token <> qr_token)
    )
  ),
  constraint reservas_retencion_pago_check check (
    (
      (
        (
          estado_reserva = 'pendiente_pago'::estado_reserva_enum
        )
        and (
          responsable_pago = 'empleado'::responsable_pago_enum
        )
        and (retencion_pago_hasta is not null)
      )
      or (
        estado_reserva <> 'pendiente_pago'::estado_reserva_enum
      )
    )
  )
) TABLESPACE pg_default;
